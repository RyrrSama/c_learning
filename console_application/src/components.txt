/**
 * @brief Sets the label for a BUTTON object by allocating or resizing memory as
 * needed.
 *
 * This function assigns a new label to the BUTTON object. If the label memory
 * is not already allocated, it allocates memory for the label. If the existing
 * memory is insufficient to hold the new label, it resizes the memory. Finally,
 * it copies the new label into the allocated memory.
 *
 * @param button Pointer to the BUTTON object whose label is to be set.
 *               The BUTTON structure must have a `label` member of type `char
 * *`.
 * @param label  Pointer to the null-terminated string to be set as the label.
 *               The function assumes the input string is valid and non-NULL.
 */
void setLabel(BUTTON *button, const char *label)
{
    // Calculate the length of the new label, including the null terminator.
    int labelLength = strlen(label) + 1;

    // If the button's label is not yet allocated, allocate memory for it.
    if (!button->label)
        button->label = (char *) malloc(labelLength * sizeof(char));
    // If the existing label memory is insufficient, resize it to fit the new
    // label.
    else if (strlen(button->label) < labelLength)
        button->label =
            (char *) realloc(button->label, labelLength * sizeof(char));

    // Copy the new label into the allocated memory, including the null
    // terminator.
    memcpy(button->label, label, labelLength);
}

/**
 * @brief Draws an anti-aliased line between two points on the console buffer.
 *
 * This function uses a modified Bresenham's line algorithm to draw a line
 * between two points (`startPoint` and `endPoint`) on the console buffer.
 * It attempts to simulate anti-aliasing by varying the pixel density along
 * the line based on the slope.
 *
 * @param consoleBuffer     Handle to the console buffer where the line will be
 * drawn.
 * @param noOfCharsWritten  A DWORD variable to store the number of characters
 * written.
 * @param startPoint        The starting coordinate of the line.
 * @param endPoint          The ending coordinate of the line.
 */

int lineAlgorithmDDA(COORD startPoint, COORD endPoint, Rect *rect)
{
    // Calculate the differences in X and Y coordinates between the start and
    // end points.
    int xDiff = endPoint.X - startPoint.X;
    int yDiff = endPoint.Y - startPoint.Y;

    // Determine the number of steps required to draw the line based on the
    // larger difference.
    int steps = abs(xDiff) > abs(yDiff) ? abs(xDiff) : abs(yDiff);
    if (rect->edges == NULL)
    {
        rect->edges = (COORD *) malloc(sizeof(COORD) * (steps + 1));
        if (rect->edges == NULL)
        {
            // Handle memory allocation failure
            return -1;
        }
    }
    else
    {
        COORD *newEdges = (COORD *) realloc(
            rect->edges, (sizeof(COORD) * (rect->edgeCount + (steps + 1))));

        if (newEdges == NULL)
        {
            // Handle memory allocation failure
            return -1;
        }
        rect->edges = newEdges;
    }

    // Calculate the increment values for X and Y coordinates per step.
    float xIncrement = xDiff / (float) steps;
    float yIncrement = yDiff / (float) steps;

    // Initialize the floating-point coordinates for the current position.
    float xCoord = startPoint.X;
    float yCoord = startPoint.Y;

    // Loop through each step to draw the line.
    for (int i = rect->edgeCount; i <= (rect->edgeCount + steps); i++)
    {
        COORD tempCoord;
        // Update the drawing point by rounding the floating-point coordinates
        // to the nearest integer.
        tempCoord.X    = (SHORT) round(xCoord);
        tempCoord.Y    = (SHORT) round(yCoord);
        rect->edges[i] = tempCoord;
        // Increment the floating-point coordinates by the calculated
        // increments.
        xCoord += xIncrement;
        yCoord += yIncrement;
    }
    rect->edgeCount += (steps + 1);
    return 0;
}
/**
 * @brief Draws a straight line between two points on the console using the
 *        Digital Differential Analyzer (DDA) algorithm.
 *
 * @param consoleBuffer A handle to the console screen buffer where the line
 *                      will be drawn.
 * @param noOfCharsWritten A DWORD variable to store the number of characters
 *                         written to the console buffer.
 * @param startPoint The starting point of the line, represented as a COORD
 *                   structure.
 * @param endPoint The ending point of the line, represented as a COORD
 *                 structure.
 * @param drawChar The character used to draw the line.
 *
 * @details This function uses the DDA algorithm to calculate intermediate
 *          points between the start and end points of the line. It then
 *          writes the specified character at each calculated point to draw
 *          the line on the console.
 *
 * @note The function assumes that the console buffer and other parameters
 *       are properly initialized before calling this function.
 */
void drawEdges(SHAPE shape)
{
    drawShapeEdges(shape.parentBuffer, shape.rect.edges, shape.rect.edgeCount,
                   shape.noOfCharsWritten, HBORDERCHAR);
}
void drawShapeEdges(HANDLE consoleBuffer, COORD edges[], DWORD edgesCount,
                    DWORD noOfCharsWritten, char drawChar)
{
    for (int index = 0; index < edgesCount; index++)
    {
        FillConsoleOutputCharacter(consoleBuffer, drawChar, 1, edges[index],
                                   &noOfCharsWritten);
    }
}

void drawLineDDA(HANDLE consoleBuffer, DWORD noOfCharsWritten, COORD startPoint,
                 COORD endPoint, char drawChar)
{
    // Initialize the current drawing point to the starting point.
    COORD drawPoint = startPoint;

    // Calculate the differences in X and Y coordinates between the start and
    // end points.
    int xDiff = endPoint.X - startPoint.X;
    int yDiff = endPoint.Y - startPoint.Y;

    // Determine the number of steps required to draw the line based on the
    // larger difference.
    int steps = abs(xDiff) > abs(yDiff) ? abs(xDiff) : abs(yDiff);

    // Calculate the increment values for X and Y coordinates per step.
    float xIncrement = xDiff / (float) steps;
    float yIncrement = yDiff / (float) steps;

    // Initialize the floating-point coordinates for the current position.
    float xCoord = startPoint.X;
    float yCoord = startPoint.Y;

    // Loop through each step to draw the line.
    for (int i = 0; i <= steps; i++)
    {
        // Draw the character at the current point on the console buffer.
        FillConsoleOutputCharacter(consoleBuffer, drawChar, 1, drawPoint,
                                   &noOfCharsWritten);

        // Increment the floating-point coordinates by the calculated
        // increments.
        xCoord += xIncrement;
        yCoord += yIncrement;

        // Update the drawing point by rounding the floating-point coordinates
        // to the nearest integer.
        drawPoint.X = round(xCoord);
        drawPoint.Y = round(yCoord);
    }
}

void drawLineCustom(HANDLE consoleBuffer, DWORD noOfCharsWritten,
                    COORD startPoint, COORD endPoint, char drawChar)
{
    // Initialize the current drawing point to the starting point.
    COORD drawPoint = startPoint;

    // Calculate the differences in X and Y coordinates between the start and
    // end points.
    int xDiff = endPoint.X - startPoint.X;
    int yDiff = endPoint.Y - startPoint.Y;

    // Adjust the differences to account for inclusive pixel drawing.
    xDiff = (xDiff > 0) ? (xDiff + 1) : (xDiff - 1);
    yDiff = (yDiff > 0) ? (yDiff + 1) : (yDiff - 1);

    // Calculate the pixel step sizes for X and Y directions.
    float xPixel = (float) abs(xDiff) / (abs(yDiff) ? abs(yDiff) : 1);
    float yPixel = (float) abs(yDiff) / (abs(xDiff) ? abs(xDiff) : 1);

    // Determine the direction of movement in X and Y axes.
    int xDirection = xDiff / (abs(xDiff) ? abs(xDiff) : 1);
    int yDirection = yDiff / (abs(yDiff) ? abs(yDiff) : 1);

    // Convert differences to absolute values for step calculation.
    xDiff = abs(xDiff);
    yDiff = abs(yDiff);

    // Determine the number of steps required to draw the line.
    int steps = (xDiff > yDiff) ? yDiff : xDiff;

    // Initialize iteration variables for fractional pixel movement.
    float yIteration = yPixel;
    float xIteration = xPixel;

    // Loop through each step to draw the line.
    for (int index = 0; index < steps; index++)
    {
        // Alternate between floor and ceil for anti-aliasing effect.
        if (index % 2 == 0)
        {
            yPixel = floor(yIteration);
            xPixel = floor(xIteration);
        }
        else
        {
            yPixel = ceil(yIteration);
            xPixel = ceil(xIteration);
        }

        // Determine whether to prioritize Y or X movement based on pixel
        // density.
        if (yPixel > xPixel)
        {
            // Draw multiple pixels in the Y direction.
            for (int yIndex = 0; yIndex < yPixel; yIndex++)
            {
                FillConsoleOutputCharacter(consoleBuffer, drawChar, 1,
                                           drawPoint, &noOfCharsWritten);
                drawPoint.Y =
                    drawPoint.Y + (1 * yDirection); // Move in Y direction.
            }
            drawPoint.X =
                drawPoint.X + (1 * xDirection); // Move one step in X direction.
        }
        else
        {
            // Draw multiple pixels in the X direction.
            for (int xIndex = 0; xIndex < xPixel; xIndex++)
            {
                FillConsoleOutputCharacter(consoleBuffer, drawChar, 1,
                                           drawPoint, &noOfCharsWritten);
                drawPoint.X =
                    drawPoint.X + (1 * xDirection); // Move in X direction.
            }
            drawPoint.Y =
                drawPoint.Y + (1 * yDirection); // Move one step in Y direction.
        }
    }
}

/**
 * @brief Draws the label of a button at its center position on the console.
 *
 * This function calculates the center position of the button's rectangle
 * and positions the console cursor accordingly. It then writes the button's
 * label at the calculated position.
 *
 * @param button The BUTTON object whose label is to be drawn.
 *               The BUTTON structure must have a `label` member of type `char
 * *` and a `shape` member of type `SHAPE`.
 */
void drawLabel(BUTTON button)
{
    // Calculate the length of the label to determine its offset for centering.
    int offset = strlen(button.label);

    // Get the center coordinate of the button's rectangle.
    COORD labelCoord = button.shape.rect.center;

    // Adjust the X-coordinate to center the label horizontally.
    labelCoord.X -= (offset / 2);

    // Set the console cursor position to the calculated label coordinate.
    SetConsoleCursorPosition(button.shape.parentBuffer, labelCoord);

    // Write the label text to the console at the calculated position.
    WriteConsole(button.shape.parentBuffer, button.label, strlen(button.label),
                 &button.shape.noOfCharsWritten, NULL);
}

/**
 * @brief Draws a custom rectangle on the specified buffer using the DDA
 * line-drawing algorithm.
 *
 * This function takes a SHAPE structure as input and draws a rectangle by
 * connecting its four corners (bottom-left, bottom-right, top-right, and
 * top-left) using the drawLineDDA function. The rectangle is drawn on the
 * parent buffer associated with the SHAPE structure.
 *
 * @param shape A SHAPE structure containing the following:
 *              - parentBuffer: The buffer where the rectangle will be drawn.
 *              - noOfCharsWritten: The number of characters written to the
 * buffer.
 *              - rect: A structure containing the coordinates of the
 * rectangle's corners:
 *                - bottomLeft: The bottom-left corner of the rectangle.
 *                - bottomRight: The bottom-right corner of the rectangle.
 *                - topRight: The top-right corner of the rectangle.
 *                - topLeft: The top-left corner of the rectangle.
 *
 * @note The function assumes that the SHAPE structure and its members are
 * properly initialized before calling this function.
 */
void drawCustomRect(SHAPE shape)
{
    // Draw the bottom edge of the rectangle
    drawLineDDA(shape.parentBuffer, shape.noOfCharsWritten,
                shape.rect.bottomLeft, shape.rect.bottomRight, HBORDERCHAR);

    // Draw the right edge of the rectangle
    drawLineDDA(shape.parentBuffer, shape.noOfCharsWritten,
                shape.rect.bottomRight, shape.rect.topRight, HBORDERCHAR);

    // Draw the top edge of the rectangle
    drawLineDDA(shape.parentBuffer, shape.noOfCharsWritten, shape.rect.topRight,
                shape.rect.topLeft, HBORDERCHAR);

    // Draw the left edge of the rectangle
    drawLineDDA(shape.parentBuffer, shape.noOfCharsWritten, shape.rect.topLeft,
                shape.rect.bottomLeft, HBORDERCHAR);
}
/**
 * @brief Draws a rectangular shape on the console using specified border
 * characters.
 *
 * This function renders a rectangle by drawing horizontal and vertical borders
 * based on the dimensions and coordinates provided in the `SHAPE` structure.
 *
 * @param shape A `SHAPE` structure containing the following:
 *              - `parentBuffer`: The handle to the console screen buffer where
 * the rectangle will be drawn.
 *              - `rect`: A structure defining the rectangle's dimensions and
 * coordinates:
 *                  - `bottomLeft`: The bottom-left corner coordinate of the
 * rectangle.
 *                  - `topLeft`: The top-left corner coordinate of the
 * rectangle.
 *                  - `width`: The width of the rectangle.
 *              - `noOfCharsWritten`: A variable to store the number of
 * characters written to the console.
 *
 * The rectangle is drawn using the following characters:
 * - Horizontal borders: `HBORDERCHAR`
 * - Vertical borders: `VBORDERCHAR`
 *
 * The function performs the following steps:
 * 1. Draws the bottom horizontal border.
 * 2. Draws the vertical borders from the bottom to the top.
 * 3. Draws the top horizontal border.
 */
void drawRect(SHAPE shape)
{
    // Draw Bottom width border
    FillConsoleOutputCharacter(shape.parentBuffer, HBORDERCHAR,
                               shape.rect.width, shape.rect.bottomLeft,
                               &shape.noOfCharsWritten);
    // Draw Vertical borders
    for (int horzontalIndex = shape.rect.bottomLeft.Y;
         horzontalIndex > shape.rect.topLeft.Y; horzontalIndex--)
    {
        COORD drawCoord = shape.rect.bottomLeft;
        drawCoord.Y     = horzontalIndex;
        FillConsoleOutputCharacter(shape.parentBuffer, VBORDERCHAR, 1,
                                   drawCoord, &shape.noOfCharsWritten);
        drawCoord.X = shape.rect.bottomLeft.X + (shape.rect.width - 1);
        FillConsoleOutputCharacter(shape.parentBuffer, VBORDERCHAR, 1,
                                   drawCoord, &shape.noOfCharsWritten);
    }
    // Draw Top width border
    FillConsoleOutputCharacter(shape.parentBuffer, HBORDERCHAR,
                               shape.rect.width, shape.rect.topLeft,
                               &shape.noOfCharsWritten);
}

/**

Rect createRect(int bottomLeftX, int bottomLeftY, int width, int height)
{
    Rect rect;
    COORD bottomLeft  = {bottomLeftX, bottomLeftY};
    COORD bottomRight = {bottomLeftX + (width - 1), bottomLeftY};
    COORD topLeft     = {bottomLeftX, bottomLeftY - (height - 1)};
    COORD topRight    = {bottomLeftX + (width - 1), bottomLeftY - (height - 1)};
    COORD center      = {bottomLeftX + (width / 2), bottomLeftY - (height / 2)};
    rect.width        = width;
    rect.height       = height;
    rect.bottomLeft   = bottomLeft;
    rect.bottomRight  = bottomRight;
    rect.topLeft      = topLeft;
    rect.topRight     = topRight;
    rect.center       = center;
    rect.edges        = NULL;
    rect.rectRegion   = NULL;
    rect.edgeCount    = 0;
    rect.regionCount  = 0;
    getShapeEdges(&rect, LINEALGORITHMDDA);
    return rect;
}

Rect createCustomRect(COORD bottomLeft, COORD bottomRight, COORD topRight,
                      COORD topLeft)
{
    Rect rect;

    rect.bottomLeft  = bottomLeft;
    rect.bottomRight = bottomRight;
    rect.topLeft     = topLeft;
    rect.topRight    = topRight;
    rect.width       = bottomRight.X - bottomLeft.X;
    rect.height      = bottomLeft.Y - topLeft.Y;
    COORD center     = {bottomLeft.X + (rect.width / 2),
                        bottomLeft.Y - (rect.height / 2)};
    rect.center      = center;
    rect.edges       = NULL;
    rect.rectRegion  = NULL;
    rect.edgeCount   = 0;
    rect.regionCount = 0;
    getShapeEdges(&rect, LINEALGORITHMDDA);
    return rect;
}

SHAPE
createRectShape(HANDLE parentBuffer, int bottomLeftX, int bottomLeftY,
                int width, int height)
{
    SHAPE shape;
    shape.rect         = createRect(bottomLeftX, bottomLeftY, width, height);
    shape.parentBuffer = parentBuffer;
    shape.draw         = drawEdges;
    return shape;
}

SHAPE
createCustomRectShape(HANDLE parentBuffer, COORD bottomLeft, COORD bottomRight,
                      COORD topRight, COORD topLeft)
{
    SHAPE shape;
    shape.rect = createCustomRect(bottomLeft, bottomRight, topRight, topLeft);
    shape.parentBuffer = parentBuffer;
    shape.draw         = drawEdges;
    return shape;
}
BUTTON createButton(HANDLE parentBuffer, const char *label, int bottomLeftX,
                    int bottomLeftY, int width, int height)
{
    BUTTON button;
    button.label = NULL;
    button.shape =
        createRectShape(parentBuffer, bottomLeftX, bottomLeftY, width, height);
    button.drawButton = drawButton;
    button.setLabel   = setLabel;
    setLabel(&button, label);
    return button;
}
